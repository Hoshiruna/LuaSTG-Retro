set(lib_name "Core.InputSystem")

set(lib_src core/InputSystem.hpp core/ControllerInput.hpp sdl/InputSystem.cpp)
if (TARGET Microsoft.GameInput)
    list(APPEND lib_src gameinput/ControllerInput.cpp)
else ()
    list(APPEND lib_src sdl/ControllerInput.cpp)
endif ()
source_group(TREE ${CMAKE_CURRENT_LIST_DIR} FILES ${lib_src})

add_library(${lib_name} STATIC)
luastg_target_common_options(${lib_name})
luastg_target_more_warning(${lib_name})
target_include_directories(${lib_name} PUBLIC .)
target_sources(${lib_name} PRIVATE ${lib_src})
target_link_libraries(${lib_name} PUBLIC SDL3::SDL3 Core.Logging Core.Math)
if (TARGET Microsoft.GameInput)
    target_compile_definitions(${lib_name} PRIVATE LUASTG_USE_GAMEINPUT NOMINMAX)
    target_link_libraries(${lib_name} PRIVATE Microsoft.GameInput)
endif ()

set_target_properties(${lib_name} PROPERTIES FOLDER engine)

if (LUASTG_BUILD_TESTS)
    set(test_name "Core.InputSystem.Test")

    add_executable(${test_name})
    luastg_target_common_options(${test_name})
    luastg_target_more_warning(${test_name})
    target_compile_features(${test_name} PRIVATE cxx_std_23)
    target_sources(${test_name} PRIVATE test/InputSystem.cpp)
    target_link_libraries(${test_name} PRIVATE options_compile_utf8 GTest::gtest_main)
    if (TARGET Microsoft.GameInput)
        # SDL virtual devices exercise the SDL controller implementation.
        target_sources(${test_name} PRIVATE sdl/InputSystem.cpp sdl/ControllerInput.cpp)
        target_include_directories(${test_name} PRIVATE .)
        target_link_libraries(${test_name} PRIVATE SDL3::SDL3 Core.Logging Core.Math)
    else ()
        target_link_libraries(${test_name} PRIVATE ${lib_name})
    endif ()

    set_target_properties(${test_name} PROPERTIES FOLDER engine/test)
endif ()
